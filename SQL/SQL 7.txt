--eg 1
--declare all required variables
declare @pname varchar(20),@price int, @qty int
--declare cursor
declare prd_cursor cursor
for select productname, price, QuantityAvailable from products

--open the cursor
open prd_cursor
--read the data
fetch next from prd_cursor into @pname,@price,@qty

--iterate the cursor set as long as there are records
while @@FETCH_STATUS = 0
 begin
  print 'The Product named : ' + ' ' + @pname + ' ' + ' costs Rs : ' + cast(@price as varchar(6)) +
   ' and the Availabilty is : ' + cast(@qty as varchar(5))
   --start to fetch again
fetch next from prd_cursor into @pname,@price,@qty
end
--close the cursor
close prd_cursor
--deallocate the cursor
deallocate prd_cursor

--eg 2 curosr with tsql
declare @sal decimal, @ename varchar(30)
declare empcursor cursor
for select empname, salary from tblemployee
open empcursor
fetch next from empcursor into @ename,@sal
 while @@FETCH_STATUS = 0
  begin
   if(@sal > 7000)
     begin 
	 print @ename + ' Earns ' + cast(@sal as varchar(6)) 
	 end
	 else
	  begin
	  print @ename + ' needs Increment '
	  end
	  fetch next from empcursor into @ename, @sal
	  end
	  close empcursor
 deallocate empcursor

declare @c1 char, @c2 char, @c3 char

 case  
   when @c1 is not null then @c1
   when @c2 is null then ' is a null value'
   end as 'CaseResult'

--cube eg
select coalesce(departmentid, 100) as Department, coalesce(gender,'All Genders'),
sum(salary) as 'Total Salary' from tblemployee
group by cube(departmentid, gender)
order by departmentid, gender

select coalesce(d.deptname, 'All Departments') as Department, coalesce(e.gender,'All Genders')as Gender,
sum(e.salary) as 'Total Salary' from tblemployee e, tbldepartment d
where e.departmentid=d.deptid
group by cube(d.deptname, e.gender)
order by d.deptname, e.gender
