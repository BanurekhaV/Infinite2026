--_ [underscore] - substitutes exactly one character

-- ^[] [Caret] - substitutes characters that are not within the range

--display all employees who work in dept as that of 103 and earns salary > 105

--this involves 2 subqueries
select Empname, Salary, Departmentid from tblemployee where departmentid =  
                       (select departmentid from tblemployee where empid = 103)
			   and Salary > 
					   (select salary from tblemployee where empid=105)

-- find employee names and salary whose salary >= to the minimum salary
select empname, salary from tblemployee where salary >=
                            (select min(salary) from tblemployee)

--list dept wise minimum salary that are > than the minimum salary of dept 4
select departmentid, min(salary) as 'Min Salary' from tblemployee
group by DepartmentId
having min(salary) > (select min(salary) as Minimum from tblemployee where DepartmentId = 3)
