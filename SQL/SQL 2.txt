--wild card characters with like
--% [percentile] - substitution of 0,1,or more characters
--_ [underscore] - substitutes exactly one character
--[range] - substitutes characters within the given range
-- ^[] [Caret] - substitutes characters that are not within the range

--get all employees where empname starts with 'a' or ends with 'a'
select * from tblemployee where empname like 'a%'
select * from tblemployee where empname like '%a'
select * from tblemployee where empname like '%h%'

--a,r,o as their second character
select * from tblemployee where empname like '_[aro]%'
select * from tblemployee where empname like '_[^aro]%'
select * from tblemployee where empname like '__[bo]%'
select * from tblemployee where empname like '__[^bo]%'
select * from tblemployee where empname like '[akg]%'
select * from tblemployee where empname like '[akg]_v%'

--sorting

--multiple column sort
