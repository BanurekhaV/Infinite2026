--views with no check option
create or alter view empview
as select empid,empname,salary from tblemployee where empid < 112

select * from empview

insert into empview values(207,'Praveena',5550)
update empview set salary = 5700 where empid=206  -- successfully executed, but no modification in the base table
                                          -- as the row is out of scope in the view (<112)
  update empview set salary = 5900 where empid=102   -- succesful with modification as the view carries 103 (<112)

create or alter view empview
as select empid,empname,salary from tblemployee where empid < 112
with check option

insert into empview values(208,'Prabhas',6000)   -- fails as it is checked with the view definition (empid <112)


--Triggers
create or alter trigger trgEmpInsert
on tblemployee   --object/entity on which the trigger is placed
for insert      -- event on which the trigger would automatically fire
as
 begin
 print 'Trigger trgEmpInsert is fired'
 select * from inserted
 select * from tblEmployee
end

-- the above trigger will be called automatically by th engine
-- whenever we insert records into the tblemployee table

  create table AuditLog(msg nvarchar(max))

  create or alter trigger trgAudit
  on tblemployee
  for insert 
  as
  begin
   declare @id int
   select @id = empid from inserted

   --dml ops inside trigger body
   insert into AuditLog(msg)
   values('New Employee with Employee id : ' + cast(@id as varchar(5)) +
   ' ' + ' is added at ' + cast(getdate() as varchar(20)))
   end

   --verify the above trigger
