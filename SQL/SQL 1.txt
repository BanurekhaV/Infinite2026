--top selects
select top 3 * from tblemployee

select top 13 percent * from tblemployee

select top 2 empname, salary from tblemployee

--relational operators
select empname from tblemployee where salary >= 5700

select empname from tblemployee where salary <> 5700

--logical operators
select empname, salary from tblemployee where salary >= 5600 and salary <=6000

select empname, DepartmentId from tblemployee where DepartmentId <5 and Salary > 5600

--between operator
select empname, salary from tblemployee where salary between 5600 and 6000

--not between
select empname, salary from tblemployee where salary not between 5600 and 6000

--null values
select empname, salary from tblemployee where DepartmentId is null

select empname, salary from tblemployee where DepartmentId is not null

--in / not in
select empname, salary , departmentid from tblemployee where DepartmentId in(1,2,4)

select empname, salary , departmentid from tblemployee where DepartmentId not in(1,2)

--arithmetic operators
select empname 'Employee Name',Salary, (Salary*12) as 'Yearly Salary' from tblEmployee

select empname, salary , (Salary +1000) 'Revised Salary'  
from tblemployee where DepartmentId in(1,2,4)

--usage of functions with select clause
