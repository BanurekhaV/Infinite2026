 as(select 0,DATENAME(dw,0)
  union all
 select n + 1, datename(dw,n+1) from cte_week where n < 6)

 --main query

 with monthcte as(select 1 as 'MonthNumber',
  Datename(month, cast('2026-01-01' as Date)) as 'MonthName'
   union all
   select MonthNumber + 1, datename(month,dateadd(month,MonthNumber,'2026-01-01'))
     from monthcte where monthnumber <12)
--main query

with ctr(initialvalue) as
(select 1 union all select initialvalue + 1 from ctr
where initialvalue <10)
