--Hierachial queries using recursive ctes:
 with hcte(EmpNo,EName,Manager,Emplevel)
  as(select empno,ename,[mgr-id],1 EmpLevel
   from emp where [mgr-id] is null      --- initial or first level / anchor member
   union all
   select e.empno,e.ename,e.[mgr-id], oq.emplevel  + 1  --  recursive value
    from emp e join hcte oq on e.[mgr-id] = oq.empno where e.[mgr-id] is not null)

	--main query
	select * from hcte
	order by emplevel

with Destination as
(select Routeid,sourcecityid,destinationcityid,cast(cityname as varchar(max))as Route,
distance from cityRoute cr join cityData cd on cr.Sourcecityid = cd.Id

union all
select cr.routeid, cr.sourcecityid,cr.destinationcityid,
cast(R.Route + '->'  + cd.cityname as varchar(max)),
 r.distance + cr.distance from Destination R
 join cityRoute cr on r.DestinationcityId = cr.Sourcecityid
